cmake_minimum_required(VERSION 3.25)
project(rife_superboost VERSION 0.1.0 LANGUAGES CXX CUDA)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The TensorRT backend requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
find_package(CUDAToolkit 13.2 REQUIRED)
set(TENSORRT_ROOT "" CACHE PATH "Official TensorRT 10.16.1.11 SDK")
file(READ "${TENSORRT_ROOT}/include/NvInferVersion.h" TRT_VERSION_HEADER)
foreach(PART MAJOR MINOR PATCH BUILD)
  string(REGEX MATCH "#define TRT_${PART}_ENTERPRISE +([0-9]+)" MATCH "${TRT_VERSION_HEADER}")
  set(TRT_${PART} "${CMAKE_MATCH_1}")
endforeach()
if(NOT "${TRT_MAJOR}.${TRT_MINOR}.${TRT_PATCH}.${TRT_BUILD}" STREQUAL "10.16.1.11")
  message(FATAL_ERROR "An exact TensorRT 10.16.1.11 SDK is required")
endif()
foreach(LIB nvinfer nvonnxparser)
  add_library(TRT_${LIB} SHARED IMPORTED)
  set_target_properties(TRT_${LIB} PROPERTIES
    IMPORTED_IMPLIB "${TENSORRT_ROOT}/lib/${LIB}_10.lib"
    IMPORTED_LOCATION "${TENSORRT_ROOT}/bin/${LIB}_10.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${TENSORRT_ROOT}/include")
endforeach()
add_library(rsb_fusions OBJECT fusion_plugin.cpp fusion.cu)
target_link_libraries(rsb_fusions PRIVATE TRT_nvinfer CUDA::cudart_static)
add_library(rsb_runtime SHARED runtime.cpp transfer.cu $<TARGET_OBJECTS:rsb_fusions>)
target_link_libraries(rsb_runtime PRIVATE TRT_nvinfer CUDA::cudart_static)
add_executable(rsb_builder builder.cpp $<TARGET_OBJECTS:rsb_fusions>)
target_link_libraries(rsb_builder PRIVATE TRT_nvinfer TRT_nvonnxparser CUDA::cudart_static)
add_executable(rsb_progress progress.cpp)
foreach(TARGET rsb_runtime rsb_builder rsb_progress rsb_fusions)
  target_compile_definitions(${TARGET} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${TARGET} PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>
    $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W3,/utf-8,/MT>)
  set_target_properties(${TARGET} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
